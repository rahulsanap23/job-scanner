  (function () {
    const originalQuerySelector = Document.prototype.querySelector;
    const originalQuerySelectorAll = Document.prototype.querySelectorAll;
    const originalSetTimeout = window.setTimeout;

    // Only intercept the selectors this locale-switcher logic actually uses
    const scopedSelectors = ['.navbar_locale-item', '.w-locales-item', '.navbar_locale-wrapper', '.navbar_aux-link'];

    let activeRoot = null; // fallback root used inside replayed setTimeout callbacks

    function shouldScope(selector) {
      return typeof selector === 'string' && scopedSelectors.some((s) => selector.indexOf(s) !== -1);
    }

    function currentNavbarRoot() {
      const script = document.currentScript;
      return script ? script.closest('.navbar_component') : null;
    }

    Document.prototype.querySelector = function (selector) {
      if (shouldScope(selector)) {
        const root = currentNavbarRoot() || activeRoot;
        if (root) return root.querySelector(selector);
      }
      return originalQuerySelector.call(this, selector);
    };

    Document.prototype.querySelectorAll = function (selector) {
      if (shouldScope(selector)) {
        const root = currentNavbarRoot() || activeRoot;
        if (root) return root.querySelectorAll(selector);
      }
      return originalQuerySelectorAll.call(this, selector);
    };

    window.setTimeout = function (fn, delay, ...args) {
      const root = currentNavbarRoot();
      if (typeof fn === 'function' && root) {
        const wrapped = function () {
          const prevActive = activeRoot;
          activeRoot = root;
          try {
            return fn.apply(this, arguments);
          } finally {
            activeRoot = prevActive;
          }
        };
        return originalSetTimeout.call(window, wrapped, delay, ...args);
      }
      return originalSetTimeout.call(window, fn, delay, ...args);
    };
  })();

<script>
  (function () {
    'use strict';

    /*
     * Prevent duplicate initialization if this code is evaluated
     * more than once.
     */
    if (window.__mizNavInit) return;
    window.__mizNavInit = true;

    /*
     * ============================================================
     * SELECTORS AND SETTINGS
     * ============================================================
     */

    var MENU = '.global-nav_main-menu';

    // Custom nested submenu elements.
    var ROOT = '.global-nav_dropdown-sub-menu-dropdown';
    var WRAP = '.global-nav_sub-menu-toggle-wrap';
    var COL = '.global-nav_dropdown-column';
    var LIST = '.global-nav_dropdown-sub-menu-dropdown-list';

    // Custom nested submenu open state.
    var OPEN = 'is-open';
    var OPENSEL = ROOT + '.' + OPEN;

    // Delay before closing a nested submenu on desktop.
    var DELAY = 250;

    /*
     * These are the three link classes currently used inside the
     * navbar.
     */
    var LINK_SELECTOR = [MENU + ' a.global-nav_dropdown-toggle-link[href]', MENU + ' a.global-nav_dropdown-sub-menu-dropdown-toggle-link[href]', MENU + ' a.global-nav_dropdown-link[href]'].join(', ');

    /*
     * Determine the input mode once on initialization.
     */
    var touch = window.matchMedia('(hover: none)').matches || 'ontouchstart' in window;

    /*
     * Desktop navigation breakpoint.
     */
    function wide() {
      return window.matchMedia('(min-width: 992px)').matches;
    }

    /*
     * querySelectorAll helper.
     */
    function qa(selector, context) {
      return (context || document).querySelectorAll(selector);
    }

    /*
     * ============================================================
     * NESTED SUBMENU HELPERS
     * ============================================================
     */

    /*
     * Clear a pending nested-submenu close timer.
     */
    function clearTimer(element) {
      if (!element || !element.__mizNavCloseTimer) return;

      clearTimeout(element.__mizNavCloseTimer);
      element.__mizNavCloseTimer = null;
    }

    /*
     * Check whether an element is inside the global navigation.
     */
    function isInsideNav(element) {
      return !!(element && element.closest && element.closest(MENU));
    }

    /*
     * Find the parent nested submenu.
     */
    function getParentSubmenu(element) {
      if (!element || !element.parentElement) return null;

      return element.parentElement.closest(ROOT);
    }

    /*
     * Open the selected nested submenu and all parent nested
     * submenus.
     */
    function openSubmenu(element) {
      var current = element;

      while (current) {
        clearTimer(current);
        current.classList.add(OPEN);
        current = getParentSubmenu(current);
      }

      syncColumnHeights();
    }

    /*
     * Close the selected nested submenu and every open submenu
     * nested inside it.
     */
    function closeSubmenu(element) {
      if (!element) return;

      clearTimer(element);
      element.classList.remove(OPEN);

      qa(OPENSEL, element).forEach(function (child) {
        clearTimer(child);
        child.classList.remove(OPEN);
      });

      syncColumnHeights();
    }

    /*
     * Schedule a nested submenu to close after the hover delay.
     */
    function scheduleSubmenuClose(element) {
      clearTimer(element);

      element.__mizNavCloseTimer = setTimeout(function () {
        closeSubmenu(element);
      }, DELAY);
    }

    /*
     * Close all nested submenus that are not the selected submenu
     * or one of its ancestors.
     */
    function closeOtherSubmenus(keep) {
      qa(OPENSEL).forEach(function (element) {
        var shouldKeep = keep && (element === keep || element.contains(keep));

        if (!shouldKeep) {
          clearTimer(element);
          element.classList.remove(OPEN);
        }
      });

      syncColumnHeights();
    }

    /*
     * Close every custom nested submenu inside a specific
     * top-level dropdown.
     */
    function closeNestedSubmenus(topDropdown) {
      if (!topDropdown) return;

      qa(ROOT, topDropdown).forEach(function (submenu) {
        clearTimer(submenu);
        submenu.classList.remove(OPEN);
      });
    }

    /*
     * ============================================================
     * TOP-LEVEL DROPDOWN HELPERS
     * ============================================================
     */

    /*
     * Remove Webflow's visual open state from a top-level dropdown.
     *
     * The forced-close CSS class remains the primary visual
     * override. Removing .w--open also resets the current Webflow
     * state as much as possible.
     */
    function clearWebflowOpenState(topDropdown) {
      if (!topDropdown) return;

      qa('.w--open', topDropdown).forEach(function (element) {
        element.classList.remove('w--open');
      });

      var toggle = topDropdown.querySelector('.w-dropdown-toggle');

      var list = topDropdown.querySelector(':scope > .global-nav_dropdown-list');

      /*
       * Fallback in case the list is not a direct child in a
       * particular navbar instance.
       */
      if (!list) {
        list = topDropdown.querySelector('.global-nav_dropdown-list');
      }

      if (toggle) {
        toggle.classList.remove('w--open');
        toggle.setAttribute('aria-expanded', 'false');
      }

      if (list) {
        list.classList.remove('w--open');
        list.style.opacity = '0';
      }
    }

    /*
     * Force-close the top-level dropdown containing the selected
     * navbar link.
     *
     * The forced-close class is intentionally not removed here.
     * It remains until physical pointer movement confirms that the
     * cursor is no longer sitting over the now-hidden dropdown.
     */
    function forceCloseTopLevelDropdown(link) {
      if (!link || !link.closest) return;

      var topDropdown = link.closest('.global-nav_dropdown');

      if (!topDropdown) return;

      closeNestedSubmenus(topDropdown);

      topDropdown.classList.add('is-nav-forced-closed');

      clearWebflowOpenState(topDropdown);
      syncColumnHeights();

      /*
       * Webflow may update its classes after the click handler has
       * completed. Clear the state again on the next animation
       * frame while the forced-close CSS remains active.
       */
      window.requestAnimationFrame(function () {
        clearWebflowOpenState(topDropdown);
      });
    }

    /*
     * Release one forced-close state.
     */
    function releaseForcedClose(topDropdown) {
      if (!topDropdown) return;

      topDropdown.classList.remove('is-nav-forced-closed');

      var list = topDropdown.querySelector(':scope > .global-nav_dropdown-list');

      if (!list) {
        list = topDropdown.querySelector('.global-nav_dropdown-list');
      }

      /*
       * Remove only the inline opacity value added by this script.
       * Webflow can then control the dropdown normally again.
       */
      if (list && list.style.opacity === '0') {
        list.style.opacity = '';
      }
    }

    /*
     * ============================================================
     * NAVBAR LINK SELECTION
     * ============================================================
     *
     * Use the click event because tabs.min.js processes the same
     * user activation to update the URL parameter and active tab.
     *
     * The listener is in the capture phase so it can detect the
     * click even if another script later stops propagation.
     *
     * The actual dropdown close runs during the next animation
     * frame, after the current click dispatch has completed.
     */

    document.addEventListener(
      'click',
      function (event) {
        if (!event.target.closest) return;

        var link = event.target.closest(LINK_SELECTOR);

        if (!link) return;

        /*
         * Leave modified clicks and links opening in a new window
         * or tab unchanged.
         */
        if (link.target === '_blank' || event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) {
          return;
        }

        window.requestAnimationFrame(function () {
          forceCloseTopLevelDropdown(link);
        });
      },
      true,
    );

    /*
     * Keyboard activation normally generates a click event, so a
     * separate Enter handler is not needed. Keeping one activation
     * pathway also prevents the close routine from firing twice.
     */

    /*
     * ============================================================
     * RELEASE THE TEMPORARY FORCED-CLOSE STATE
     * ============================================================
     *
     * Do not use mouseout here.
     *
     * Hiding the dropdown itself can produce a mouseout event,
     * which previously removed the forced-close class immediately
     * and allowed Webflow to reopen the menu.
     *
     * Instead, wait for genuine pointer movement. elementFromPoint
     * checks what is currently under the pointer after the dropdown
     * has been hidden.
     */

    document.addEventListener(
      'pointermove',
      function (event) {
        var forcedDropdowns = qa('.global-nav_dropdown.is-nav-forced-closed');

        if (!forcedDropdowns.length) return;

        var elementUnderPointer = document.elementFromPoint(event.clientX, event.clientY);

        forcedDropdowns.forEach(function (topDropdown) {
          /*
           * Keep the forced-close state if the pointer is currently
           * over the top-level dropdown's remaining visible area,
           * such as its main navigation toggle.
           */
          if (elementUnderPointer && topDropdown.contains(elementUnderPointer)) {
            return;
          }

          releaseForcedClose(topDropdown);
        });
      },
      {
        capture: true,
        passive: true,
      },
    );

    /*
     * Remove any remaining temporary forced-close state if focus
     * moves completely outside the affected dropdown.
     */
    document.addEventListener(
      'focusin',
      function (event) {
        qa('.global-nav_dropdown.is-nav-forced-closed').forEach(function (topDropdown) {
          if (!topDropdown.contains(event.target)) {
            releaseForcedClose(topDropdown);
          }
        });
      },
      true,
    );

    /*
     * ============================================================
     * DESKTOP NESTED SUBMENU BEHAVIOR
     * ============================================================
     */

    if (!touch) {
      /*
       * Open custom nested submenus on hover.
       */
      document.addEventListener('mouseover', function (event) {
        if (!isInsideNav(event.target)) return;

        var submenu = event.target.closest(ROOT);

        if (submenu) {
          openSubmenu(submenu);
        }
      });

      /*
       * Close custom nested submenus after the hover delay unless
       * the pointer moved into the same submenu or one of its
       * descendants.
       */
      document.addEventListener('mouseout', function (event) {
        if (!isInsideNav(event.target)) return;

        var destination = event.relatedTarget;
        var submenu = event.target.closest(ROOT);

        while (submenu) {
          if (destination && submenu.contains(destination)) {
            clearTimer(submenu);
          } else {
            scheduleSubmenuClose(submenu);
          }

          submenu = getParentSubmenu(submenu);
        }
      });
    } else {
      /*
       * ==========================================================
       * MOBILE AND TOUCH NESTED SUBMENU BEHAVIOR
       * ==========================================================
       */

      document.addEventListener('click', function (event) {
        if (!event.target.closest) return;

        var toggleWrap = event.target.closest(WRAP);

        if (toggleWrap && isInsideNav(toggleWrap)) {
          var submenu = toggleWrap.closest(ROOT);

          if (!submenu) return;

          /*
           * The custom submenu toggle wrapper is not itself a
           * destination link, so preventing its default behavior
           * is safe.
           */
          event.preventDefault();

          if (submenu.classList.contains(OPEN)) {
            closeSubmenu(submenu);
          } else {
            closeOtherSubmenus(submenu);
            openSubmenu(submenu);
          }

          return;
        }

        /*
         * A tap outside all custom nested submenus closes them.
         */
        if (!event.target.closest(ROOT)) {
          closeOtherSubmenus(null);
        }
      });
    }

    /*
     * ============================================================
     * DESKTOP COLUMN HEIGHT SYNCHRONIZATION
     * ============================================================
     *
     * Stretch each dropdown column to accommodate its tallest
     * currently displayed custom submenu list.
     */

    function syncColumnHeights() {
      if (!wide()) return;

      qa(MENU + ' ' + COL).forEach(function (column) {
        var tallestHeight = 0;
        var hasVisibleList = false;

        qa(LIST, column).forEach(function (list) {
          if (window.getComputedStyle(list).display !== 'none') {
            hasVisibleList = true;

            tallestHeight = Math.max(tallestHeight, list.scrollHeight);
          }
        });

        column.style.minHeight = hasVisibleList ? Math.max(tallestHeight, column.offsetHeight) + 'px' : '';
      });
    }

    /*
     * ============================================================
     * INITIAL STATE NORMALIZATION
     * ============================================================
     *
     * Remove stale custom .is-open states and Webflow component
     * variant classes that should not ship as an open state.
     */

    function normalizeNavbar() {
      try {
        qa([ROOT, ROOT + '-toggle', LIST, ROOT + '-container'].join(',')).forEach(function (element) {
          Array.prototype.slice.call(element.classList).forEach(function (className) {
            if (className.lastIndexOf('w-variant-', 0) === 0) {
              element.classList.remove(className);
            }
          });

          clearTimer(element);
          element.classList.remove(OPEN);
        });

        /*
         * Remove any temporary class left in restored or cached
         * navbar HTML.
         */
        qa('.global-nav_dropdown.is-nav-forced-closed').forEach(function (topDropdown) {
          releaseForcedClose(topDropdown);
        });
      } catch (error) {
        console.warn('[Mizuho Nav] Initialization failed:', error);
      }
    }

    function initializeNavbar() {
      normalizeNavbar();
      syncColumnHeights();
    }

    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', initializeNavbar);
    } else {
      initializeNavbar();
    }

    /*
     * ============================================================
     * RESIZE HANDLING
     * ============================================================
     */

    var resizeTimer;

    window.addEventListener('resize', function () {
      clearTimeout(resizeTimer);

      resizeTimer = setTimeout(function () {
        /*
         * Do not carry the forced desktop hover state across a
         * breakpoint change.
         */
        qa('.global-nav_dropdown.is-nav-forced-closed').forEach(function (topDropdown) {
          releaseForcedClose(topDropdown);
        });

        syncColumnHeights();
      }, 150);
    });
  })();
