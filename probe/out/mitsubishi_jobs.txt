        // Check if the file extension is allowed

        }

    function applyJobsById(){

        $('.Applyjobform').prop('disabled', true);
        $(".Applyjobform").val('Please wait...');

       var jobs_apply_by_Id = $("#jobsApplyById").val();

       if(jobs_apply_by_Id!="") {
         var job_form = document.getElementById("jobsById");
       } else {
         var job_form = document.getElementById("jobs_form");
       }
       
        var fd = new FormData(job_form);
        $.ajax({
            type:'POST',
            url: 'applyJobsByEduReq',
            data: fd,
            cache: false,
            processData: false,
            contentType: false,

            success:function(response)
            {
                data=JSON.parse(response);
                $('.Applyjobform').prop('disabled', false);
                $(".Applyjobform").val('Agree & Submit');

                if(data.status===200) {
                    if(data.id=="getId") {
                        $("#jobsById")[0].reset();
                        var scrollTop = "show_divId";
                    } else {
                        $("#jobs_form")[0].reset();
                        var scrollTop = "show_div";
                    }
                    $("#"+scrollTop).show();
                    $('.success_msg').text(data.msg);
                    document.getElementById(scrollTop).scrollIntoView();
                } else
                {
                    if(data.id=="getId") {
                        $("#jobsById")[0].reset();
                        var scrollTop = "show_error_divId";
                    } else {
                        $("#jobs_form")[0].reset();
                        var scrollTop = "show_error_div";
                    }
                    $("#"+scrollTop).show();
                    $('.error_msg').text(data.msg);
                    document.getElementById(scrollTop).scrollIntoView();
                    
                }
               
            }
        });

        return false;
    }
