        function initializeBootstrapPopovers() {
            var popoverTriggerList = document.querySelectorAll('[data-bs-toggle="popover"]');
            for (var i = 0; i < popoverTriggerList.length; i++) {
                var el = popoverTriggerList[i];
                var popover = bootstrap.Popover.getInstance(el);
                if (!popover) {
                    new bootstrap.Popover(el, {
                        trigger: 'hover focus',
                        html: true
                    });
                }
            }
        }

        $(document).ready(function () {
            initializeBootstrapPopovers();

            $(document).on('focus mouseenter', '[data-bs-toggle="popover"]', function () {
                var popover = bootstrap.Popover.getInstance(this);
                if (!popover) {
                    popover = new bootstrap.Popover(this, {
                        trigger: 'hover focus',
                        html: true
                    });
                    popover.show();
                }
            });

            if (typeof Sys !== 'undefined' && Sys.WebForms && Sys.WebForms.PageRequestManager) {
                Sys.WebForms.PageRequestManager.getInstance().add_endRequest(function () {
                    initializeBootstrapPopovers();
                });
            }

        <script>
        document.addEventListener("DOMContentLoaded", function () {

            // Remove active from all menus
            document.querySelectorAll(".navbar-nav .nav-link").forEach(function (link) {
                link.classList.remove("active");
            });

            // Get current page
            var currentPage = window.location.pathname.split('/').pop().toLowerCase();

            // If no page specified, treat as Home
            if (currentPage === "" || currentPage === "/") {
                currentPage = "index.aspx";
            }

            var activeFound = false;

            document.querySelectorAll(".navbar-nav .nav-link").forEach(function (link) {
                var href = link.getAttribute("href");

                if (href && href.toLowerCase() === currentPage) {
                    link.classList.add("active");
                    activeFound = true;
                }
            });

            // Default Home active if no match found
            if (!activeFound) {
                document.getElementById("homeSelect").classList.add("active");
            }
        });
        </script>
