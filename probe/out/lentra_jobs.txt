<!-- Cloudflare Turnstile: the widgets live inside hidden Bootstrap modals, and
     implicit auto-render is unreliable for hidden content (the token can be
     missing when the user submits -> 422 "Verification failed"). Explicitly
     render each widget when its modal opens, and reset it on close so every
     open gets a fresh token. Covers Request Demo, Contact Us and Access forms. -->
<script>
  document.addEventListener('DOMContentLoaded', function() {
    function lentraRenderTS(divId) {
      var el = document.getElementById(divId);
      if (!el || typeof turnstile === 'undefined') return;
      try {
        if (el.querySelector('iframe')) {
          turnstile.reset(el); // already rendered -> refresh token
        } else {
          turnstile.render(el, {
            sitekey: el.getAttribute('data-sitekey'),
            size: el.getAttribute('data-size') || undefined
          });
        }
      } catch (e) {
        try {
          turnstile.reset(el);
        } catch (_) {}
      }
    }
    var TS_MODALS = {
      '#requestdemo': 'ts-navdemo',
      '#exampleModalCenter': 'ts-sendmessage',
      '#requestaccess': 'ts-accessform'
    };
    Object.keys(TS_MODALS).forEach(function(modalSel) {
      var widgetId = TS_MODALS[modalSel];
      $(document).on('shown.bs.modal', modalSel, function() {
        lentraRenderTS(widgetId);
      });
      $(document).on('hidden.bs.modal', modalSel, function() {
        var el = document.getElementById(widgetId);
        if (el && typeof turnstile !== 'undefined' && el.querySelector('iframe')) {
          try {
            turnstile.reset(el);
          } catch (e) {}
        }
      });
    });
  });
</script>
