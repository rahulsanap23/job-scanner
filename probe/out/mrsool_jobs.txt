    function updateVideo(index) {
        if (backgroundVideo.src !== videos[index]) {
            backgroundVideo.pause(); // Stop the current video

            backgroundVideo.load(); // Reload the video
            backgroundVideo.play().catch((err) => {

            }); // Play the new video
        }
